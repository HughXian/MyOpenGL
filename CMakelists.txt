# -----------全局输出路径
# 函数：自动转换vs平台名称 win32 -> x86
function(set_vs_output_dir target_name)
    if(MSVC)
        #获取 当前sln平台
        if(CMAKE_VS_PLATFORM_NAME STREQUAL "WIN32")
            set(PLATFORM_NAME "X86")
        elseif(CMAKE_VS_PLATFORM_NAME STREQUAL "X64")
            set(PLATFORM_NAME "X64")
        else()
            set(PLATFORM_NAME ${CMAKE_VS_PLATFORM_NAME})
        endif()

        #拼接路径
        set(RUNTIME_OUTPUT  "${CMAKE_SOURCE_DIR}/bin/${PLATFORM_NAME}/$<CONFIG>")
        set(ARCHIVE_OUT     "${CMAKE_BINARY_DIR}/lib/${PLATFORM_NAME}/$<CONFIG>")
        set(PDB_OUT         "${CMAKE_SOURCE_DIR}/bin/${PLATFORM_NAME}/$<CONFIG>")

        # 给目标单独设置输出目录
        set_target_properties(${target_name} PROPERTIES
            RUNTIME_OUTPUT_DIRECTORY    ${RUNTIME_OUTPUT}
            LIBRARY_OUTPUT_DIRECTORY    ${RUNTIME_OUTPUT}
            ARCHIVE_OUTPUT_DIRECTORY    ${ARCHIVE_OUT}
            PDB_OUTPUT_DIRECTORY        ${PDB_OUT}
            )

    endif()

endfunction()

# 自动搜集当前目录源码
file(GLOB_RECURSE MAIN_SOURCES CONFIGURE_DEPENDS
    LIST_DIRECTORIES FALSE
    "${CMAKE_CURRENT_LIST_DIR}/*.h"
    "${CMAKE_CURRENT_LIST_DIR}/*.hpp"
    "${CMAKE_CURRENT_LIST_DIR}/*.cpp"
)

add_executable(OpenGLMain ${MAIN_SOURCES})

# VS筛选器：自动映射磁盘真实目录结构
source_group(
    TREE ${CMAKE_SOURCE_DIR}/src
    PREFIX ""
    FILES ${MAIN_SOURCES}
)

# 设置输出目录
set_vs_output_dir(OpenGLMain)

# 头文件路径
target_include_directories(OpenGLMain PRIVATE
    ${GLAD_ROOT}/include
    ${GLFW_ROOT}/include
    # 如果后续依赖自己写的模块：
    # ${CMAKE_SOURCE_DIR}/src/GameCore
)

# 链接外部库
target_link_libraries(OpenGLMain PRIVATE
    glfw
    # 后续自己写的业务静态库在这里追加
    # GameCore
    # Render
)

# Windows系统依赖
